<script>
function switchView(viewClass){
  document.body.className = viewClass;
  document.getElementById('btn-a').classList.toggle('active', viewClass === 'view-a');
  document.getElementById('btn-b').classList.toggle('active', viewClass === 'view-b');
}
function openWorkModal(title, meta){
  const modal = document.getElementById('work-modal');
  document.getElementById('modal-title').textContent = title;
  document.getElementById('modal-meta').textContent = meta;
  const bm = document.getElementById('modal-bm');
  bm.classList.remove('saved');
  bm.textContent = 'Listeme Ekle';
  modal.showModal();
}
function toggleBookmark(btn){
  const saved = btn.classList.toggle('saved');
  btn.textContent = saved ? 'Kaydedildi ✓' : 'Listeme Ekle';
}
